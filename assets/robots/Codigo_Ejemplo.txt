#include <NewPing.h>

// 1. Pines de Motores (L298N)
const int IN1 = 5; const int IN2 = 6; 
const int IN3 = 9; const int IN4 = 10; 

// 2. Sensores Ultrasonidos
#define MAX_DIST 100 
NewPing us_Frontal(2, 4, MAX_DIST);
NewPing us_Derecho(7, 8, MAX_DIST);
NewPing us_Izquierdo(12, 13, MAX_DIST);

// 3. Parámetros de Control
int velBase = 110;      // Velocidad crucero moderada para evitar derrapes
float Kp = 1.5;         // Fuerza de corrección
float Kd = 12.0;        // Amortiguación para evitar el zigzag
int errorAnterior = 0;

void setup() {
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  Serial.begin(115200);
}

void loop() {
  // Lectura de distancias (en cm)
  int d_der = us_Derecho.ping_cm();
  int d_izq = us_Izquierdo.ping_cm();
  int d_frt = us_Frontal.ping_cm();

  // Si el sensor no recibe eco (0), asumimos que la pared está lejos (MAX_DIST)
  if (d_der == 0) d_der = MAX_DIST;
  if (d_izq == 0) d_izq = MAX_DIST;
  if (d_frt == 0) d_frt = MAX_DIST;

  // --- LÓGICA DE PASILLO ---
  
  // Calculamos el error de centrado
  int errorActual = d_izq - d_der; 

  // Si detectamos pared al frente muy cerca, reducimos velocidad base para girar mejor
  int vBaseActual = velBase;
  if (d_frt < 20) {
    vBaseActual = 70; // Frenado preventivo en curvas cerradas
  }

  // PD: Proporcional + Derivativo
  float correccion = (errorActual * Kp) + ((errorActual - errorAnterior) * Kd);
  
  // Aplicamos a los motores
  int vI = vBaseActual - correccion;
  int vD = vBaseActual + correccion;

  aplicarMotores(vI, vD);
  
  errorAnterior = errorActual;
}

void aplicarMotores(int vL, int vR) {
  // IMPORTANTE: En el simulador, valores negativos causan inversión brusca.
  // Mantenemos los motores entre 0 y 255 para giros suaves.
  vL = constrain(vL, 0, 255);
  vR = constrain(vR, 0, 255);

  // Motor Izquierdo
  analogWrite(IN1, vL);
  digitalWrite(IN2, LOW);

  // Motor Derecho
  analogWrite(IN3, vR);
  digitalWrite(IN4, LOW);
}
